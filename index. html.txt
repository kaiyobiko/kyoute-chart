<script>
let myChart = null;

function generateChart() {
    // 各フォームから数値を取得（空欄なら0点にする）
    const kokugo = parseFloat(document.getElementById('kokugo').value) || 0;
    const su1a = parseFloat(document.getElementById('su1a').value) || 0;
    const su2bc = parseFloat(document.getElementById('su2bc').value) || 0;
    const eigoR = parseFloat(document.getElementById('eigoR').value) || 0;
    const eigoL = parseFloat(document.getElementById('eigoL').value) || 0;

    // グラフを表示するエリアを可視化
    document.getElementById('chartArea').style.display = 'block';

    // 既にグラフが作られていたら一度破棄する（再入力に対応するため）
    if (myChart !== null) {
        myChart.destroy();
    }

    // Chart.jsで円グラフを描画
    const ctx = document.getElementById('myPieChart').getContext('2d');
    myChart = new Chart(ctx, {
        type: 'pie',
        data: {
            labels: ['国語', '数学ⅠA', '数学ⅡBC', '英語R', '英語L'],
            datasets: [{
                data: [kokugo, su1a, su2bc, eigoR, eigoL],
                backgroundColor: [
                    '#e74c3c', // 国語：赤
                    '#3498db', // 数学1A：青
                    '#2980b9', // 数学2BC：濃い青
                    '#2ecc71', // 英語R：緑
                    '#27ae60'  // 英語L：濃い緑
                ],
                borderWidth: 1
            }]
        },
        options: {
            responsive: true,
            plugins: {
                legend: { position: 'bottom' },
                title: { display: true, text: '得点バランス（比率）' }
            }
        }
    });
    
    // グラフの位置まで自動スクロール
    document.getElementById('chartArea').scrollIntoView({ behavior: 'smooth' });
}
</script>
